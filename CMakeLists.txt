cmake_minimum_required(VERSION 3.20)
project(kunpeng_tls VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KUNPENG_BUILD_ITRUSTEE "Build AArch64 iTrustee TA and Linux CA" OFF)
option(KUNPENG_SIGN_TA "Sign TA with SDK and caller private configuration" OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build")
endif()
set(ITRUSTEE_SDK "${PROJECT_SOURCE_DIR}/third_party/itrustee_sdk" CACHE PATH "iTrustee SDK")
if(KUNPENG_BUILD_ITRUSTEE)
  include(cmake/openssl.cmake)
else()
  find_package(OpenSSL 3.5 REQUIRED COMPONENTS SSL Crypto)
endif()
add_library(tls_engine STATIC src/tls_engine.c)
target_include_directories(tls_engine PUBLIC include)
target_link_libraries(tls_engine PUBLIC OpenSSL::SSL OpenSSL::Crypto)
target_compile_features(tls_engine PUBLIC c_std_11)
target_compile_options(tls_engine PRIVATE -Wall -Wextra -Werror)
find_package(Threads REQUIRED)
add_library(tls_host_runtime STATIC ca/RestAPIHost.cpp)
target_include_directories(tls_host_runtime PUBLIC ca include)
target_link_libraries(tls_host_runtime PUBLIC Threads::Threads)
target_compile_features(tls_host_runtime PUBLIC cxx_std_17)
target_compile_options(tls_host_runtime PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
add_executable(tls_ca ca/tls_host.cpp ca/ITrusteeTlsSessionService.cpp
  "${ITRUSTEE_SDK}/src/CA/libteec_adaptor.c")
set_target_properties(tls_ca PROPERTIES OUTPUT_NAME kunpeng-tls-host)
target_include_directories(tls_ca PRIVATE include "${ITRUSTEE_SDK}/include/CA")
target_link_libraries(tls_ca PRIVATE tls_host_runtime ${CMAKE_DL_LIBS})
target_compile_options(tls_ca PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
message(STATUS "Host CLI uses real SDK adaptor; running TLS requires iTrustee and /usr/lib64/libteec.so")
if(KUNPENG_BUILD_ITRUSTEE)
  include(cmake/itrustee.cmake)
else()
  if(KUNPENG_SIGN_TA)
    message(FATAL_ERROR "KUNPENG_SIGN_TA requires KUNPENG_BUILD_ITRUSTEE")
  endif()
  enable_testing()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/ta/manifest.txt")
  file(STRINGS "${PROJECT_SOURCE_DIR}/ta/manifest.txt" heap_budget_lines
    REGEX "^[ \t]*gpd\\.ta\\.dataSize[ \t]*:")
  list(LENGTH heap_budget_lines heap_budget_count)
  if(NOT heap_budget_count EQUAL 1 OR
      NOT heap_budget_lines MATCHES "^[ \t]*gpd\\.ta\\.dataSize[ \t]*:[ \t]*([1-9][0-9]*)[ \t]*$")
    message(FATAL_ERROR "Manifest must contain one positive decimal dataSize")
  endif()
  set(heap_budget "${CMAKE_MATCH_1}")
  add_executable(heap_tracker_test tests/heap_tracker_test.c tests/heap_tracker.c)
  target_compile_features(heap_tracker_test PRIVATE c_std_11)
  target_compile_options(heap_tracker_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME heap_tracker_test COMMAND heap_tracker_test)
  add_executable(tls_heap tests/tls_heap.c tests/heap_tracker.c tests/heap_engine.c)
  target_include_directories(tls_heap PRIVATE include)
  target_compile_features(tls_heap PRIVATE c_std_11)
  target_compile_options(tls_heap PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(tls_heap PRIVATE OpenSSL::SSL OpenSSL::Crypto)
  add_test(NAME tls_heap COMMAND tls_heap "${heap_budget}")
  add_test(NAME tls_heap_budget_failure COMMAND "${CMAKE_COMMAND}"
    "-DPROGRAM=$<TARGET_FILE:tls_heap>" -P "${PROJECT_SOURCE_DIR}/tests/heap_budget_failure.cmake")
  set_tests_properties(tls_heap tls_heap_budget_failure PROPERTIES TIMEOUT 180)
  add_executable(tls_integration tests/tls_integration.c)
  target_link_libraries(tls_integration PRIVATE tls_engine OpenSSL::SSL OpenSSL::Crypto)
  target_compile_options(tls_integration PRIVATE -Wall -Wextra -Werror)
  add_test(NAME tls_integration COMMAND tls_integration)
  set_tests_properties(tls_integration PROPERTIES TIMEOUT 30)
  add_executable(tls_backpressure tests/tls_integration.c src/tls_engine.c)
  target_include_directories(tls_backpressure PRIVATE include)
  target_compile_definitions(tls_backpressure PRIVATE TLS_ENGINE_TEST_BIO_CAPACITY=64)
  target_compile_options(tls_backpressure PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(tls_backpressure PRIVATE OpenSSL::SSL OpenSSL::Crypto)
  add_test(NAME tls_backpressure COMMAND tls_backpressure)
  set_tests_properties(tls_backpressure PROPERTIES TIMEOUT 30)
  add_executable(host_concurrency_test tests/host_concurrency_test.cpp)
  target_link_libraries(host_concurrency_test PRIVATE tls_host_runtime)
  target_compile_options(host_concurrency_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME host_concurrency_test COMMAND host_concurrency_test)
  set_tests_properties(host_concurrency_test PROPERTIES TIMEOUT 30)
  add_executable(itrustee_service_test tests/itrustee_service_test.cpp ca/ITrusteeTlsSessionService.cpp)
  target_include_directories(itrustee_service_test PRIVATE ca include "${ITRUSTEE_SDK}/include/CA")
  target_link_libraries(itrustee_service_test PRIVATE Threads::Threads)
  target_compile_options(itrustee_service_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME itrustee_service_test COMMAND itrustee_service_test)
  set_tests_properties(itrustee_service_test PROPERTIES TIMEOUT 30)
  add_executable(host_tls_test tests/host_tls_test.cpp)
  target_link_libraries(host_tls_test PRIVATE tls_host_runtime tls_engine)
  target_compile_options(host_tls_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME host_tls_test COMMAND host_tls_test)
  set_tests_properties(host_tls_test PROPERTIES TIMEOUT 30)
  add_executable(ta_sessions_test tests/ta_sessions_test.c)
  target_include_directories(ta_sessions_test PRIVATE tests include)
  target_link_libraries(ta_sessions_test PRIVATE Threads::Threads)
  target_compile_options(ta_sessions_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME ta_sessions_test COMMAND ta_sessions_test)
  set_tests_properties(ta_sessions_test PROPERTIES TIMEOUT 30)
endif()
